cmake_minimum_required(VERSION 3.18)
project(MIOpenSmokeTests LANGUAGES CXX)

enable_testing()
option(TEST_AI_MODELS "Test the bundled AI models" ON)
option(TEST_CONVOLUTION "Test FP32 and FP16 convolutions" ON)

if(TEST_AI_MODELS)
    find_package(Eigen3 REQUIRED)
    find_package(nlohmann_json CONFIG REQUIRED)
    if(NOT TARGET nlohmann_json)
        add_library(nlohmann_json ALIAS nlohmann_json::nlohmann_json)
    endif()
    find_package(frugally-deep 0.15.31 EXACT CONFIG REQUIRED)
    add_executable(test_ai_models test_ai_models.cpp)
    target_compile_features(test_ai_models PRIVATE cxx_std_17)
    target_link_libraries(test_ai_models PRIVATE
        frugally-deep::fdeep Eigen3::Eigen nlohmann_json::nlohmann_json)
    find_path(MIOPEN_MODEL_DIR gfx908.tn.model PATH_SUFFIXES share/miopen/db REQUIRED)
    add_test(NAME ai_models COMMAND test_ai_models "${MIOPEN_MODEL_DIR}")
endif()

if(TEST_CONVOLUTION)
    find_package(hip CONFIG REQUIRED)
    find_path(MIOPEN_INCLUDE_DIR miopen/miopen.h REQUIRED)
    find_library(MIOPEN_LIBRARY MIOpen REQUIRED)
    add_executable(test_convolution test_convolution.cpp)
    target_compile_features(test_convolution PRIVATE cxx_std_17)
    target_include_directories(test_convolution PRIVATE "${MIOPEN_INCLUDE_DIR}")
    target_link_libraries(test_convolution PRIVATE hip::host "${MIOPEN_LIBRARY}")
    add_test(NAME convolution COMMAND bash
        "${CMAKE_CURRENT_SOURCE_DIR}/test_convolution.sh" $<TARGET_FILE:test_convolution>)
    set_tests_properties(convolution PROPERTIES SKIP_RETURN_CODE 77)
endif()
